V6_DIV_SEG_MAX=3

V6_DIV_COLO_MAX=9
# 快筛候选抽取（按机场分组采样）：从全部延迟候选里，每个机场（colo）取延迟最低的

# 0 = 关闭分组采样，退回纯延迟顺位（"延迟最优优先"模式，会牺牲机场多样性）
V6_COLO_PICK=6

# ---- 池构建 ----

# 每轮入池的段总数（学习段 + 随机段）；段越多探索面越大
V6_SEG_PICK=20
# 随机段探索去重窗口：随机抽样时避开最近抽中的这么多条段（滑动窗口，段被抽中后约停留
# 本值/每轮随机段量 轮后重新可抽），扩大探索覆盖、避免连续几轮重复探索同一批段；
# 0 = 关闭去重（退回纯随机）。默认 48 ≈ 4 轮随机段量（段源不足时自动回退允许重抽保量）
V6_AVOID_KEEP=48
# 学习段封顶数量（防止历史好段过多挤占随机探索名额）
V6_LEARN_MAX=8
# 学习段滑动窗口：段学习只统计最近这么多份历史 best（时间倒序），过时段自动滑出学习池，
# 紧跟 CF 优选变化——避免历史累计次数把已过时段"软固化"在池里。24 份 ≈ 1.3 天（每天约 19 轮）；
# 调小更灵敏（学习样本变少），调大更稳（惯性更强）；0 = 不限制（统计全部历史，退回旧行为）
V6_LEARN_WINDOW=24
# 每条入池段展开的段内随机 IPv6 数：脚本自己从段生成随机 IP 入池（候选量精确可控、全段均匀采样），
# 不依赖 cfst 的段内生成（随机游走量不可控且分布偏向段头）；
# 候选量 ≈ SEG_PICK x N + 定向 IP + 外部 IP + 自建池；0 = 关闭展开（退回整段 CIDR 交给 cfst 随机游走）
V6_IPS_PER_SEG=200
# 本地优选段（V6url.txt 手写 CIDR）每条段展开的 IP 数：手写段 = 人工优选段，已确认为好段，
# 无需像订阅段那样 200 个撒网，展开少量即可；且手写段"确定性全进"，段数多时靠此值控制总量
V6_LOCAL_IPS_PER_SEG=254
# 学习定向：自建池每个历史好 IP 的 /64 邻域生成 N 个随机 IP（同 /64 同地区，定点覆盖好 IP
# 所在的优质邻域，比 /48 全段随机更容易命中）；0 = 关闭
# 注：v4 无对应项——/24 全段仅 254 个地址，IPS_PER_SEG=200 已近穷举，定向无意义
V6_TGT_PER_IP=25
# 学习定向 /64 前缀数量上限：自建池好 IP 的 /64 去重后最多取这么多邻域（上限 = 本值 x TGT_PER_IP 个）
V6_TGT_MAX_PREFIX=16
# 外部 IP 抽样配额：V6url.txt 中若出现纯 IP 行（非 CIDR 段，如别人优选好的 IP 列表），
# 全部源合并去重后随机抽这么多条直接入池测速（cfst 按 /128 原样测）；0 = 忽略外部 IP
V6_IP_PICK=100
# 自建池入池标准：最近一轮 best 的上榜 IP（延迟<SELF_LAT）直接复测（连续性保障）；
# 学习段标准：同一段前缀（v6 /48、v4 /24）累计"上榜轮次"下限（同一段一轮最多计 1 次，
# 避免单轮刷满 10 条污染学习）——随机生成下 IP 不会重复，只有"段"会跨轮重复命中，故学习以段为单位
V6_SELF_MIN=2
# 自建池/学习判定延迟上限（毫秒），超过此延迟的历史 IP 不参与复测与学习
# 【调优】250 → 200：PDX/SEA/LAX（200~215ms）实测中位只有 0.55~0.72MB/s，
# 进榜纯占名额；200 仍留得住 SJC（约 186ms，实测 1.4~2.5MB/s）
V6_SELF_LAT=200

# ---- 稳定性与超时 ----

# 历史归档保留份数（48 份 ≈ 每天约 19 轮 × 2 天）
V6_KEEP=48
# 延迟测速失败（429 等）后的最大重扫轮数，防止夜间死转
V6_MAX_SCAN_ROUNDS=30
# 429 指数退避开关：1=重试/重探间隔逐次翻倍（BASE→BASE*2→BASE*4，封顶 BACKOFF_CAP） 0=固定 BACKOFF_BASE 秒
V6_BACKOFF_ENABLE=1
# 429 退避基础间隔（秒）：第 1 次等待该值，之后逐次翻倍
V6_BACKOFF_BASE=120
# 429 退避封顶（秒）：翻倍到此值后不再增加（探测等待用；下载重试最多到 BASE*4）
V6_BACKOFF_CAP=600
# 429 持续等待上限（秒）：累计等待超过此值仍未解除则放弃本轮，防止无限挂起拖垮后续轮次
# 【调优】1200 → 600：单轮实测只需 200~340s，等 20 分钟已远超收益，且会与下一个 cron 撞车
V6_429_WAIT_MAX=600
# 单轮整体超时（秒）：总调度用 timeout 强制结束超时的子脚本（退出码 124）。
# 【调优】2400 → 1200（20 分钟）：正常一轮 200~340s；2400 会让"卡住的那轮"跨过下一个 cron
# （11-03 21:32 那轮就是这样把总调度锁留到 22:17，吞掉一整轮）
V6_TIMEUP=1200

# ---- 测速参数 ----

# 延迟测速并发线程数（cfst -n）：Go 协程做 TLS 握手，48 线程对路由器压力很小；
# 若发现延迟测量虚高、频繁 429 或 cfst 启动即崩（内存不足），可降到 24~32
V6_LAT_THREAD=48
# 第一段（粗扫）每个 IP 的探测次数：1 = 只探一次，只为快速剔掉死 IP / 高延迟 IP
# 注：该参数原名"延迟测速单 IP 超时"，实际对应 cfst 的 -t（探测次数），不是超时
V6_LAT_TIMEOUT=1
# 第二段（精测延迟）每个 IP 的探测次数（cfst -t）：4 次取平均才压得住单次探测的抖动；
# 两段式总代价约为"全量 4 次"的 1/3（第二段只测第一段的幸存者，约 2400 个）
V6_LAT_TRYS=4
# 第二段丢包上限（cfst -tlr，0.00~1.00）：4 次里丢 1 次 = 0.25 > 0.2 即淘汰；
# 老版本没设该项（默认 1.00），会把不稳定的 IP 留在候选里
V6_LAT_LOSS=0.2
# 单次下载限时（秒）：优质 CF IP 下载 9.1MB 只需 3~4 秒，10 秒上限等于 ~0.9MB/s 以下直接判废
V6_DL_MAXTIME=10
# 单次下载失败（429 / 超时 / 5xx）的重试上限（超过则该样本记失败并剔除，不再写 0 分进榜）
V6_DL_RETRY=3
# 精测每个候选的下载次数（取平均定最终速度）
# 【调优】1 → 2：实测同一 IP 连续 3 次 9.1MB 的稳态速度可差 10~20%（3.20 / 2.16 / 3.01 MB/s），
# 单样本无法区分 2.0 与 2.6；2 次取平均可把噪声压到 ~7%。
# 流量代价：FAST_TOPN x DL_RUNS x 9.1MB（16 x 2 ≈ 290MB/轮，已按流量折中取值）
V6_DL_RUNS=2
# 快筛幸存者的定序方式：lat = 延迟优先（默认，配合机场/段配额取多样性）
#                        speed = 按快筛速度优先（旧行为；1MB 样本与 9MB 精测相关性仅 ~0.7，不建议）
V6_PICK_MODE=lat
# cfst 启动期内存不足（"out of memory allocating heap arena map"）时降到的并发数（重试 1 次）
V6_OOM_THREAD=24
# 达标速度阈值（MB/s）：仅用于完成日志标注"达到 X MB/s 标准"，不影响测速与排序
# 【调优】8 → 3.0：历史最高 4.07MB/s（06~11 点是全天最好时段），8 永远不触发
V6_SPEED_PASS=3.0

# 429 持续等待上限（秒）：累计等待超过此值仍未解除则放弃本轮，防止无限挂起拖垮后续轮次
# 【调优】1200 → 600（同 v6）
V4_429_WAIT_MAX=600
# 单轮整体超时（秒）：总调度用 timeout 强制结束超时的子脚本（退出码 124）
# 【调优】2400 → 1200（同 v6：正常一轮 200~340s，避免挂起轮次跨过下一个 cron）
V4_TIMEUP=1200

# ---- 测速参数 ----

# 延迟测速并发线程数（cfst -n）：同 v6，48 线程提速约 1/3，最高建议 64；内存紧张时降到 24~32
V4_LAT_THREAD=48
# 第一段（粗扫）每个 IP 的探测次数（cfst -t，非超时）
V4_LAT_TIMEOUT=1
# 第二段（精测延迟）每个 IP 的探测次数（cfst -t）：同 v6 两段式
V4_LAT_TRYS=4
# 第二段丢包上限（cfst -tlr）：同 v6
V4_LAT_LOSS=0.2
# 单次下载限时（秒）：同 v6，10 秒 ≈ 0.9MB/s 以下判废
V4_DL_MAXTIME=10
# 单次下载失败（429 / 超时 / 5xx）的重试上限（超过则剔除该样本，不再记 0 分进榜）
V4_DL_RETRY=3
# 精测每个候选的下载次数（取平均定最终速度）：【调优】1 → 2，同 v6（单样本噪声 10~20%）
V4_DL_RUNS=2
# 快筛幸存者定序：lat = 延迟优先（默认）/ speed = 快筛速度优先（旧行为）
V4_PICK_MODE=lat
# cfst 启动期内存不足时降到的并发数（重试 1 次）
V4_OOM_THREAD=24
# 达标速度阈值（MB/s）：仅用于日志标注
# 【调优】8 → 3.0：v4 的历史最高 3.62MB/s，8 永远不触发
V4_SPEED_PASS=3.0

V4_LAT_URL="https://cp.cloudflare.com/generate_204"

V4_FAST_URL="https://cdnjs.cloudflare.com/ajax/libs/echarts/5.4.3/echarts.min.js"
# 精测下载源（约 9.1MB，jsDelivr 的 Cloudflare 专用子域）
V4_DL_URL="https://testingcf.jsdelivr.net/npm/typescript@5.4.5/lib/typescript.js"
