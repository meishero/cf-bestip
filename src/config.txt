# ==================== IPv6 测速配置 ====================
# 分组：结果与筛选 / 池构建 / 稳定性与超时 / 测速参数 / 测速目标（与 IPv4 段逐项对齐）

# ---- 结果与筛选 ----

# 最终排名保留名次（结果文件与 GitHub 榜单条数）
V6_TOPN=15
# 结果文件名（本地生成 + GitHub 同名上传）：移植到不同宽带运营商的路由器时，只改这里即可整套复用——
# 移动宽带用 best_ip_v6_yidong.txt、电信改 best_ip_v6_dianxin.txt；多台路由器可共用一个 GitHub 仓库互不覆盖
V6_RESULT_FILE=""
# 延迟测速后进入快筛的名次
V6_DL_TOPN=45
# 快筛后进入精测的名次（建议与 V6_TOPN 对齐）
V6_FAST_TOPN=15
# 上榜多样性：同一 /48 段最多上榜条数（防止榜单被单一 /48 包揽、段故障时全军覆没；不足时按速度顺位回填）
V6_DIV_SEG_MAX=2
# 上榜多样性：同地区（机场码，如 HKG/LAX/MAA）最多上榜条数（不足时按速度顺位回填）
V6_DIV_COLO_MAX=7
# 快筛候选抽取（按机场分组采样）：从全部延迟候选里，每个机场（colo）取延迟最低的
# COLO_PICK 个进快筛，强制每个机场都有代表（防单一机场延迟优势霸榜，实测曾出现
# 30 个快筛候选 100% 同机场）；不足 DL_TOPN 时按延迟顺位回填。
# 0 = 关闭分组采样，退回纯延迟顺位（"延迟最优优先"模式，会牺牲机场多样性）
V6_COLO_PICK=5

# ---- 池构建 ----

# 每轮入池的段总数（学习段 + 随机段）；段越多探索面越大
V6_SEG_PICK=20
# 随机段探索去重窗口：随机抽样时避开最近抽中的这么多条段（滑动窗口，段被抽中后约停留
# 本值/每轮随机段量 轮后重新可抽），扩大探索覆盖、避免连续几轮重复探索同一批段；
# 0 = 关闭去重（退回纯随机）。默认 48 ≈ 4 轮随机段量（段源不足时自动回退允许重抽保量）
V6_AVOID_KEEP=48
# 学习段封顶数量（防止历史好段过多挤占随机探索名额）
V6_LEARN_MAX=8
# 学习段滑动窗口：段学习只统计最近这么多份历史 best（时间倒序），过时段自动滑出学习池，
# 紧跟 CF 优选变化——避免历史累计次数把已过时段"软固化"在池里。24 份 ≈ 1.3 天（每天约 19 轮）；
# 调小更灵敏（学习样本变少），调大更稳（惯性更强）；0 = 不限制（统计全部历史，退回旧行为）
V6_LEARN_WINDOW=24
# 每条入池段展开的段内随机 IPv6 数：脚本自己从段生成随机 IP 入池（候选量精确可控、全段均匀采样），
# 不依赖 cfst 的段内生成（随机游走量不可控且分布偏向段头）；
# 候选量 ≈ SEG_PICK x N + 定向 IP + 外部 IP + 自建池；0 = 关闭展开（退回整段 CIDR 交给 cfst 随机游走）
V6_IPS_PER_SEG=200
# 本地优选段（V6url.txt 手写 CIDR）每条段展开的 IP 数：手写段 = 人工优选段，已确认为好段，
# 无需像订阅段那样 200 个撒网，展开少量即可；且手写段"确定性全进"，段数多时靠此值控制总量
V6_LOCAL_IPS_PER_SEG=20
# 学习定向：自建池每个历史好 IP 的 /64 邻域生成 N 个随机 IP（同 /64 同地区，定点覆盖好 IP
# 所在的优质邻域，比 /48 全段随机更容易命中）；0 = 关闭
# 注：v4 无对应项——/24 全段仅 254 个地址，IPS_PER_SEG=200 已近穷举，定向无意义
V6_TGT_PER_IP=25
# 学习定向 /64 前缀数量上限：自建池好 IP 的 /64 去重后最多取这么多邻域（上限 = 本值 x TGT_PER_IP 个）
V6_TGT_MAX_PREFIX=16
# 外部 IP 抽样配额：V6url.txt 中若出现纯 IP 行（非 CIDR 段，如别人优选好的 IP 列表），
# 全部源合并去重后随机抽这么多条直接入池测速（cfst 按 /128 原样测）；0 = 忽略外部 IP
V6_IP_PICK=100
# 自建池入池标准：最近一轮 best 的上榜 IP（延迟<SELF_LAT）直接复测（连续性保障）；
# 学习段标准：同一段前缀（v6 /48、v4 /24）累计"上榜轮次"下限（同一段一轮最多计 1 次，
# 避免单轮刷满 10 条污染学习）——随机生成下 IP 不会重复，只有"段"会跨轮重复命中，故学习以段为单位
V6_SELF_MIN=2
# 自建池/学习判定延迟上限（毫秒），超过此延迟的历史 IP 不参与复测与学习
# v6 直连 CF 普遍 <250ms，250 可兼顾质量与候选量
V6_SELF_LAT=250

# ---- 稳定性与超时 ----

# 历史归档保留份数（48 份 ≈ 每天约 19 轮 × 2 天）
V6_KEEP=48
# 延迟测速失败（429 等）后的最大重扫轮数，防止夜间死转
V6_MAX_SCAN_ROUNDS=30
# 429 指数退避开关：1=重试/重探间隔逐次翻倍（BASE→BASE*2→BASE*4，封顶 BACKOFF_CAP） 0=固定 BACKOFF_BASE 秒
V6_BACKOFF_ENABLE=1
# 429 退避基础间隔（秒）：第 1 次等待该值，之后逐次翻倍
V6_BACKOFF_BASE=120
# 429 退避封顶（秒）：翻倍到此值后不再增加（探测等待用；下载重试最多到 BASE*4）
V6_BACKOFF_CAP=600
# 429 持续等待上限（秒）：累计等待超过此值仍未解除则放弃本轮，防止无限挂起拖垮后续轮次
V6_429_WAIT_MAX=1200
# 单轮整体超时（秒）：总调度用 timeout 强制结束超时的子脚本（退出码 124）。
# 2400 = 40 分钟：给 429 退避（WAIT_MAX=1200s）+ 延迟测速 + 下载留足余量，避免正常慢轮被误杀；
# 每轮正常约 11 分钟，远低于此；仍低于 cron 每小时一次的间隔
V6_TIMEUP=2400

# ---- 测速参数 ----

# 延迟测速并发线程数（cfst -n）：Go 协程做 TLS 握手，48 线程对路由器压力很小，
# 比默认 32 提速约 1/3；若发现延迟测量虚高或频繁 429 可降回 32
V6_LAT_THREAD=48
# 延迟测速单 IP 超时（秒，cfst 最低 1）：HTTPing 计时含 TCP+TLS 握手全程，
# 响应超过 1 秒的 IP 必然超过 -tl 延迟门槛（v6 250 / v4 300），提前判死零误杀，
# 可把死 IP 的扫描等待时间从 2s 降到 1s（候选多时整轮省好几分钟）
V6_LAT_TIMEOUT=1
# 单次下载限时（秒）：优质 CF IP 下载 9.1MB 只需 1~3 秒，10 秒上限等于
# ~0.9MB/s 以下直接判废（这种 IP 最终也不会用），避免慢 IP 拖满 30 秒浪费整轮时间
V6_DL_MAXTIME=10
# 单次下载撞 429 的重试上限（超过则该样本记 0 分跳过）
V6_DL_RETRY=3
# 精测每个候选的下载次数（取平均定最终速度）：2 次是"噪声抑制 vs 流量/耗时"的平衡；
# 降到 1 可省约一半精测流量（约 146MB/轮）与时间，速度测量噪声略增（低带宽场景几乎无感）；
# 提到 3+ 仅在高带宽（>4MB/s）节点测量抖动明显时才有价值
V6_DL_RUNS=1
# 达标速度阈值（MB/s）：仅用于完成日志标注"达到 X MB/s 标准"，不影响测速与排序
V6_SPEED_PASS=8

# ---- 测速目标 ----

# HTTPing 延迟测速目标（CF captive portal 检测域，无 bot 防护基本不限流；勿用 www.cloudflare.com，有 WAF 拦 Go 客户端）
V6_LAT_URL="https://cp.cloudflare.com/generate_204"
# 快筛下载源（约 1MB，cdnjs echarts.min.js，公共 CDN、CF 边缘接入，可 --resolve 强制 IP；
# 快筛只做粗排，1MB 足够分辨快慢且流量比 3.6MB 省 2/3）
V6_FAST_URL="https://cdnjs.cloudflare.com/ajax/libs/echarts/5.4.3/echarts.min.js"
# 精测下载源（约 9.1MB，jsDelivr 的 Cloudflare 专用子域；勿用 cdn.jsdelivr.net 主域，已迁 Fastly 无法强制 CF IP）
V6_DL_URL="https://testingcf.jsdelivr.net/npm/typescript@5.4.5/lib/typescript.js"
